/-! A 15-coordinate connected CSS example. Only `Std` is imported.
All inputs are seven-bit supports on physical coordinates 0,...,6.
The actual decoding graph has 15 distinct X-check rows. -/

abbrev Word := BitVec 15
abbrev Input := BitVec 7
abbrev V := Fin 15
abbrev ZC := Fin 7
abbrev Msg := V (*@\ensuremath{\to}@*) V (*@\ensuremath{\to}@*) Bool

def xs : List Word := [213,102,179,120,173,30,203,10880,13056,6528,15360,5760,3840,9600,16512]
def zs : List Word := [85,102,120,127,27265,13056,15360]
def xrow (c : V) : Word := xs[c.val]!
def zrow (d : ZC) : Word := zs[d.val]!
def vertices : List V := List.finRange 15

def bit (w : Word) (v : V) : Bool := w.getLsbD v.val
def parity (bs : List Bool) := bs.foldl Bool.xor false
def dot (a b : Word) : Bool := parity (vertices.map fun v => bit a v && bit b v)
def pack (f : V (*@\ensuremath{\to}@*) Bool) : Word :=
  BitVec.ofNat 15 ((vertices.map fun v => if f v then 2^v.val else 0).foldl ((*@\ensuremath{\cdot}@*)+(*@\ensuremath{\cdot}@*)) 0)
def embed (e : Input) : Word := BitVec.ofNat 15 e.toNat
def support : Word := 127
def adj (c v : V) : Bool := bit (xrow c) v

def syndrome (e : Word) : Word := pack fun c => dot (xrow c) e

def majority (bs : List Bool) : Bool := decide (bs.length < 2 * bs.countP id)
def checkUpdate (s : Word) (x : Msg) : Msg := fun c v =>
  adj c v && Bool.xor (bit s c)
    (parity (vertices.map fun u => adj c u && (u != v) && x c u))
def hard (y : Msg) : Word :=
  pack fun v => majority ((vertices.filter fun c => adj c v).map fun c => y c v)
def variableUpdate (y : Msg) : Msg := fun c v =>
  adj c v && majority ((vertices.filter fun d => adj d v && (d != c)).map fun d => y d v)
def edgeSyndrome (x : Msg) : Word :=
  pack fun c => parity (vertices.map fun v => adj c v && x c v)
def firstChecks (e : Input) : Msg := checkUpdate (syndrome (embed e)) (fun _ _ => false)
def firstHard (e : Input) : Word := hard (firstChecks e)
def firstEdges (e : Input) : Msg := variableUpdate (firstChecks e)
def terminal (e : Input) : Word := if syndrome (embed e) = 0 then 0 else firstHard e

def Absorbing (a : Word) : Prop := (*@\ensuremath{\forall}@*) v : V, bit a v = true (*@\ensuremath{\to}@*)
  (vertices.filter fun c => adj c v && dot (xrow c) a).length <
  (vertices.filter fun c => adj c v && !(dot (xrow c) a)).length

theorem selected_absorbing : Absorbing support := by
  unfold Absorbing
  decide

theorem induced_checks_even : (*@\ensuremath{\forall}@*) c : V, dot (xrow c) support = false := by decide

theorem selected_size : (vertices.filter fun v => bit support v).length = 7 := by decide

theorem distinct_x_checks : xs.Pairwise (fun a b => a (*@\ensuremath{\ne}@*) b) := by decide

theorem css_commutation : (*@\ensuremath{\forall}@*) c : V, (*@\ensuremath{\forall}@*) d : ZC, dot (xrow c) (zrow d) = false := by decide

theorem first_hard_matches : (*@\ensuremath{\forall}@*) e : Input,
  syndrome (firstHard e) = syndrome (embed e) := by decide

theorem first_edge_matches : (*@\ensuremath{\forall}@*) e : Input,
  edgeSyndrome (firstEdges e) = syndrome (embed e) := by decide

theorem terminal_matches : (*@\ensuremath{\forall}@*) e : Input,
  syndrome (terminal e) = syndrome (embed e) := by decide

theorem terminal_equals_first : (*@\ensuremath{\forall}@*) e : Input, terminal e = firstHard e := by decide

def zCombination (a : BitVec 7) : Word :=
  ((List.finRange 7).map fun d => if a.getLsbD d.val then zrow d else 0).foldl ((*@\ensuremath{\cdot}@*) ^^^ (*@\ensuremath{\cdot}@*)) 0

def xBasis : List Word := [213,102,120,10880,13056,15360,16512]

  ((List.finRange 7).map fun d => if a.getLsbD d.val then xBasis[d.val]! else 0).foldl ((*@\ensuremath{\cdot}@*) ^^^ (*@\ensuremath{\cdot}@*)) 0

-- Actual bipartite connectivity, including every check and every variable.
abbrev Node := Sum V V

def Edge : Node (*@\ensuremath{\to}@*) Node (*@\ensuremath{\to}@*) Prop
  | .inl v, .inr c => adj c v = true
  | .inr c, .inl v => adj c v = true
  | _, _ => False

inductive Walk : Node (*@\ensuremath{\to}@*) Node (*@\ensuremath{\to}@*) Prop
  | nil (u : Node) : Walk u u
  | cons {u v w : Node} : Edge u v (*@\ensuremath{\to}@*) Walk v w (*@\ensuremath{\to}@*) Walk u w

theorem edge_symm {u v : Node} (h : Edge u v) : Edge v u := by
  cases u <;> cases v <;> simp_all [Edge]

theorem walk_trans {u v w : Node} (h : Walk u v) (k : Walk v w) : Walk u w := by
  induction h with
  | nil => exact k
  | cons h _ ih => exact Walk.cons h (ih k)

theorem walk_reverse {u v : Node} (h : Walk u v) : Walk v u := by
  induction h with
  | nil u => exact Walk.nil u
  | @cons u v w h hv ih =>
    exact walk_trans ih (Walk.cons (edge_symm h) (Walk.nil u))

theorem hub_incidence : (*@\ensuremath{\forall}@*) v : V, (*@\ensuremath{\exists}@*) c : V,
  adj c v = true (*@\ensuremath{\land}@*) adj c 7 = true := by decide

theorem checks_nonempty : (*@\ensuremath{\forall}@*) c : V, (*@\ensuremath{\exists}@*) v : V, adj c v = true := by decide

theorem variable_to_hub (v : V) : Walk (.inl v) (.inl 7) := by
  obtain (*@\ensuremath{\langle}@*)c,hv,h7(*@\ensuremath{\rangle}@*) := hub_incidence v
  exact Walk.cons (v := .inr c) hv (Walk.cons (v := .inl 7) h7 (Walk.nil (.inl 7)))

theorem node_to_hub (u : Node) : Walk u (.inl 7) := by
  cases u with
  | inl v => exact variable_to_hub v
  | inr c =>
    obtain (*@\ensuremath{\langle}@*)v,hv(*@\ensuremath{\rangle}@*) := checks_nonempty c
    exact Walk.cons (v := .inl v) hv (variable_to_hub v)

theorem decoding_graph_connected (u v : Node) : Walk u v := by
  exact walk_trans (node_to_hub u) (walk_reverse (node_to_hub v))

  cut.getLsbD 0 (*@\ensuremath{\ne}@*) cut.getLsbD 7 (*@\ensuremath{\lor}@*)
  cut.getLsbD 0 (*@\ensuremath{\ne}@*) cut.getLsbD 4 (*@\ensuremath{\lor}@*)
  cut.getLsbD 0 (*@\ensuremath{\ne}@*) cut.getLsbD 2 (*@\ensuremath{\lor}@*)

  cut.getLsbD 7 (*@\ensuremath{\ne}@*) cut.getLsbD 11 (*@\ensuremath{\lor}@*)
  cut.getLsbD 7 (*@\ensuremath{\ne}@*) cut.getLsbD 9 (*@\ensuremath{\lor}@*)
  cut.getLsbD 7 (*@\ensuremath{\ne}@*) cut.getLsbD 13 (*@\ensuremath{\lor}@*)
  cut.getLsbD 7 (*@\ensuremath{\ne}@*) cut.getLsbD 14 (*@\ensuremath{\lor}@*)
  cut.getLsbD 4 (*@\ensuremath{\ne}@*) cut.getLsbD 5 (*@\ensuremath{\lor}@*)
  cut.getLsbD 4 (*@\ensuremath{\ne}@*) cut.getLsbD 3 (*@\ensuremath{\lor}@*)
  cut.getLsbD 2 (*@\ensuremath{\ne}@*) cut.getLsbD 1 (*@\ensuremath{\lor}@*)
  cut.getLsbD 11 (*@\ensuremath{\ne}@*) cut.getLsbD 12 (*@\ensuremath{\lor}@*)
  cut.getLsbD 11 (*@\ensuremath{\ne}@*) cut.getLsbD 10 (*@\ensuremath{\lor}@*)
  cut.getLsbD 9 (*@\ensuremath{\ne}@*) cut.getLsbD 8

theorem tree_crosses (cut : Word) (h0 : cut (*@\ensuremath{\ne}@*) 0) (h1 : cut (*@\ensuremath{\ne}@*) 32767) : TreeCrosses cut := by
  apply Classical.byContradiction
  intro hn
  simp only [TreeCrosses,not_or,Classical.not_not] at hn
  rcases hn with (*@\ensuremath{\langle}@*)he0,he1,he2,he3,he4,he5,he6,he7,he8,he9,he10,he11,he12,he13(*@\ensuremath{\rangle}@*)
  have hv0 : cut.getLsbD 0 = cut.getLsbD 0 := rfl
  have hv7 : cut.getLsbD 7 = cut.getLsbD 0 := he0.symm.trans hv0
  have hv4 : cut.getLsbD 4 = cut.getLsbD 0 := he1.symm.trans hv0
  have hv2 : cut.getLsbD 2 = cut.getLsbD 0 := he2.symm.trans hv0
  have hv6 : cut.getLsbD 6 = cut.getLsbD 0 := he3.symm.trans hv0
  have hv11 : cut.getLsbD 11 = cut.getLsbD 0 := he4.symm.trans hv7
  have hv9 : cut.getLsbD 9 = cut.getLsbD 0 := he5.symm.trans hv7
  have hv13 : cut.getLsbD 13 = cut.getLsbD 0 := he6.symm.trans hv7
  have hv14 : cut.getLsbD 14 = cut.getLsbD 0 := he7.symm.trans hv7
  have hv5 : cut.getLsbD 5 = cut.getLsbD 0 := he8.symm.trans hv4
  have hv3 : cut.getLsbD 3 = cut.getLsbD 0 := he9.symm.trans hv4
  have hv1 : cut.getLsbD 1 = cut.getLsbD 0 := he10.symm.trans hv2
  have hv12 : cut.getLsbD 12 = cut.getLsbD 0 := he11.symm.trans hv11
  have hv10 : cut.getLsbD 10 = cut.getLsbD 0 := he12.symm.trans hv11
  have hv8 : cut.getLsbD 8 = cut.getLsbD 0 := he13.symm.trans hv9
  have hall (i : Nat) (hi : i < 15) : cut.getLsbD i = cut.getLsbD 0 := by
    have hcases : i = 0 (*@\ensuremath{\lor}@*) i = 1 (*@\ensuremath{\lor}@*) i = 2 (*@\ensuremath{\lor}@*) i = 3 (*@\ensuremath{\lor}@*) i = 4 (*@\ensuremath{\lor}@*) i = 5 (*@\ensuremath{\lor}@*) i = 6 (*@\ensuremath{\lor}@*) i = 7 (*@\ensuremath{\lor}@*) i = 8 (*@\ensuremath{\lor}@*) i = 9 (*@\ensuremath{\lor}@*) i = 10 (*@\ensuremath{\lor}@*) i = 11 (*@\ensuremath{\lor}@*) i = 12 (*@\ensuremath{\lor}@*) i = 13 (*@\ensuremath{\lor}@*) i = 14 := by omega
    rcases hcases with hi | hi | hi | hi | hi | hi | hi | hi | hi | hi | hi | hi | hi | hi | hi
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv0
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv1
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv2
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv3
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv4
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv5
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv6
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv7
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv8
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv9
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv10
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv11
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv12
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv13
    (*@\ensuremath{\cdot}@*) simpa [hi] using hv14
  cases hroot : cut.getLsbD 0 with
  | false =>
    apply h0
    apply BitVec.eq_of_getLsbD_eq
    intro i hi
    rw [hall i hi,hroot]
    simp
  | true =>
    have hc : cut = BitVec.allOnes 15 := by
      apply BitVec.eq_of_getLsbD_eq
      intro i hi
      calc cut.getLsbD i = true := (hall i hi).trans hroot
           _ = (BitVec.allOnes 15).getLsbD i := by
             rw [BitVec.getLsbD_allOnes]
             simp only [hi, decide_true]

def CSSStabilizer (p : Word (*@\ensuremath{\times}@*) Word) : Prop := XSpan p.1 (*@\ensuremath{\land}@*) ZSpan p.2

def SplitsAcross (cut : Word) : Prop :=
  (*@\ensuremath{\forall}@*) p : Word (*@\ensuremath{\times}@*) Word, CSSStabilizer p (*@\ensuremath{\to}@*) CSSStabilizer (p.1 &&& cut, p.2 &&& cut)

theorem no_physical_product_cut (cut : Word) (h0 : cut (*@\ensuremath{\ne}@*) 0) (h1 : cut (*@\ensuremath{\ne}@*) 32767) :
  (*@\ensuremath{\neg}@*) SplitsAcross cut := by
  intro hs
  obtain (*@\ensuremath{\langle}@*)i,hi(*@\ensuremath{\rangle}@*) := cut_anticommutation cut h0 h1
  obtain (*@\ensuremath{\langle}@*)a,ha(*@\ensuremath{\rangle}@*) := actual_x_rows_in_basis_span (tx i)
  have hp : CSSStabilizer (xrow (tx i), 0) := (*@\ensuremath{\langle}@*)(*@\ensuremath{\langle}@*)a,ha(*@\ensuremath{\rangle}@*),zero_z_span(*@\ensuremath{\rangle}@*)
  obtain (*@\ensuremath{\langle}@*)b,hb(*@\ensuremath{\rangle}@*) := (hs _ hp).1
  have hc := span_x_commutes b (tz i)
  rw [hb] at hc
  rw [hi] at hc
  contradiction

-- Eventual properties refer to the specified decoder with syndrome stopping.
def stoppedOutput (e : Input) (t : Nat) : Word := if t = 0 then 0 else terminal e

def EventuallyCorrect (e : Input) (c : V) : Prop :=
  (*@\ensuremath{\exists}@*) L : Nat, (*@\ensuremath{\forall}@*) t : Nat, L (*@\ensuremath{\le}@*) t (*@\ensuremath{\to}@*)
    bit (syndrome (stoppedOutput e t)) c = bit (syndrome (embed e)) c

def EventuallyConverges (e : Input) (v : V) : Prop :=
  (*@\ensuremath{\exists}@*) L : Nat, (*@\ensuremath{\forall}@*) t : Nat, L (*@\ensuremath{\le}@*) t (*@\ensuremath{\to}@*) bit (stoppedOutput e t) v = bit (stoppedOutput e (t+1)) v

def TrappingSelected : Prop := (*@\ensuremath{\exists}@*) e : Input,
  ((*@\ensuremath{\exists}@*) c : V, (*@\ensuremath{\neg}@*) EventuallyCorrect e c) (*@\ensuremath{\lor}@*)
  ((*@\ensuremath{\exists}@*) v : V, bit support v = true (*@\ensuremath{\land}@*) (*@\ensuremath{\neg}@*) EventuallyConverges e v)

theorem all_checks_eventually_correct (e : Input) (c : V) : EventuallyCorrect e c := by
  refine (*@\ensuremath{\langle}@*)1,?_(*@\ensuremath{\rangle}@*)
  intro t ht
  have h : t (*@\ensuremath{\ne}@*) 0 := by omega
  simp only [stoppedOutput,ite_eq_right h]
  rw [terminal_matches]

theorem all_variables_converge (e : Input) (v : V) : EventuallyConverges e v := by
  refine (*@\ensuremath{\langle}@*)1,?_(*@\ensuremath{\rangle}@*)
  intro t ht
  have h : t (*@\ensuremath{\ne}@*) 0 := by omega
  have h' : t+1 (*@\ensuremath{\ne}@*) 0 := by omega
  simp only [stoppedOutput,ite_eq_right h,ite_eq_right h']

theorem selected_not_trapping : (*@\ensuremath{\neg}@*) TrappingSelected := by
  intro (*@\ensuremath{\langle}@*)e,h(*@\ensuremath{\rangle}@*)
  rcases h with (*@\ensuremath{\langle}@*)c,hc(*@\ensuremath{\rangle}@*) | (*@\ensuremath{\langle}@*)v,hv,hf(*@\ensuremath{\rangle}@*)
  (*@\ensuremath{\cdot}@*) exact hc (all_checks_eventually_correct e c)
  (*@\ensuremath{\cdot}@*) exact hf (all_variables_converge e v)

def cnotX (c t : V) (w : Word) : Word :=
  if bit w c then w ^^^ BitVec.ofNat 15 (2^t.val) else w
def cnotZ (c t : V) (w : Word) : Word :=
  if bit w t then w ^^^ BitVec.ofNat 15 (2^c.val) else w
