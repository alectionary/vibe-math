theorem nonzero_weight (x : Fin 3 (*@$\to$@*) ZMod L (*@$\to$@*) ZMod 2) (hx : Kernel x) (hn : x (*@$\ne$@*) 0) :
    (CoverWeight.support (word x)).card = 2*L := by
  have heven := (kernel_iff x).mp hx |>.2
  have hne : x 0 0 (*@$\ne$@*) 0 (*@$\lor$@*) x 1 0 (*@$\ne$@*) 0 (*@$\lor$@*) x 2 0 (*@$\ne$@*) 0 := by
    by_contra h
    push Not at h
    apply hn
    funext i s
    rw [all_constant x hx i s]
    fin_cases i
    (*@$\cdot$@*) exact h.1
    (*@$\cdot$@*) exact h.2.1
    (*@$\cdot$@*) exact h.2.2
  have hr : (Finset.univ.filter (fun i : Fin 3 => x i 0 (*@$\ne$@*) 0)).card = 2 := by
    rw [Finset.card_eq_sum_ones,Finset.sum_filter]
    simpa [Fin.sum_univ_succ,Nat.add_assoc] using three_even_count (x 0 0) (x 1 0) (x 2 0) heven hne
  have hs : CoverWeight.support (word x) =
      (Finset.univ.filter (fun i : Fin 3 => x i 0 (*@$\ne$@*) 0)) (*@$\times$@*)(*@${}^{\mathsf s}$@*) (Finset.univ : Finset (ZMod L)) := by
    ext v
    simp [CoverWeight.support,word,all_constant x hx]
  rw [hs,Finset.card_product,hr]
  simp
