theorem source_distance_drop (hL : 11 ∣ L) :
    quantumDistance L ≤ 11 ∧ 12 ≤ classicalDistance L ∧ quantumDistance L < classicalDistance L := by
  rw [quantum_distance_eq,classical_distance_eq]
  exact ⟨DistanceFamily.quantum_distance_upper,DistanceFamily.classical_distance_lower hL,

/-- Commutation of the actual source coefficient matrices, not merely of a
finite sampled expansion. -/
theorem css_commutation (a : XCheck L) (b : ZCheck L) :
    (∑ q, HZ b q * HX a q) = 0 := by
  let w : Fin 3 → Fin 4 → ZMod L → ZMod 2 := fun i j s => if ((i,j),s)=a then 1 else 0
  have he : ∀ q, SourceModel.join (stabilizerLeft w) (stabilizerRight w) q = HX a q := by
    intro q
    cases q with
    | inl p =>
      rcases p with ⟨⟨i,j⟩,s⟩
      change stabilizerLeft w i j s = _
      rw [← x_span_left w i j s]
      simp [w]
    | inr p =>
      rcases p with ⟨⟨i,j⟩,s⟩
      change stabilizerRight w i j s = _
      rw [← x_span_right w i j s]
      simp [w]
  calc
    (∑ q, HZ b q * HX a q) =
      ∑ q, HZ b q * SourceModel.join (stabilizerLeft w) (stabilizerRight w) q := by simp only [he]
    _ = Syndrome (stabilizerLeft w) (stabilizerRight w) b.1.1 b.1.2 b.2 := z_action _ _ _ _ _
    _ = 0 := FamilyWords.css_commutation w _ _ _

CoverDistance.lean

theorem lower_bound (p : A →+ B) (e : Fin 3 → Fin 4 → A) (d s : Nat)
    (hcode : ∀ y : CyclicCover.Word B, CyclicCover.Kernel (fun i j => p (e i j)) y →
      y≠0 → d≤(CoverWeight.support y).card)
    (hstop : ∀ T : Finset (Fin 4 × B), SmallStopping.Stopping
      (CyclicCover.Adjacent (fun i j => p (e i j))) T → s≤T.card)
    (x : CyclicCover.Word A) (hx : CyclicCover.Kernel e x) (hn : x≠0) :
    min d (2*s) ≤ (CoverWeight.support x).card := by
  let y := CoverWeight.push (CyclicCover.project p) x
  by_cases hy : y=0
  · have hs := hstop _ (CyclicCover.projected_support_stopping p e x hx hn)
    have hw := CoverWeight.zero_push_weight_doubles (CyclicCover.project p) x hy
    exact (Nat.min_le_right d (2*s)).trans ((Nat.mul_le_mul_left 2 hs).trans hw)
  · have hd := hcode y (CyclicCover.kernel_push p e x hx) hy
    have hw := CoverWeight.push_weight_le (CyclicCover.project p) x
    change (CoverWeight.support y).card ≤ (CoverWeight.support x).card at hw
    exact (Nat.min_le_left d (2*s)).trans (hd.trans hw)
